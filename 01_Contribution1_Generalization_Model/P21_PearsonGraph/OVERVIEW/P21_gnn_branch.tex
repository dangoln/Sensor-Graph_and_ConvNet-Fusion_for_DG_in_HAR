% =====================================================================

%  Needs the PlotNeuralNet layer macros, exactly like the example:
%      ./layers/init.sty  (provides the Box={...} pic)
%  Compile:  pdflatex P21_gnn_branch.tex   (or lualatex)
%
%  Forward spine:
%    Raw window
%      ├─ (2A) Conv1D blk1[CNN1D|BN|ReLU] -> Conv1D blk2[CNN1D|BN|ReLU]
%      │       -> AvgPool -> Linear -> Node emb
%      └─ (2B) Pearson rho -> Scoring -> Top-k -> Edge_index
%    (Node emb + Edge_index) -> GraphSAGE x2 [Dropout|SAGEConv|BN|ReLU]
%    -> Global Mean Pool
%      ├─ Activity Linear -> Softmax -> P_GNN
%      └─ Domain head [GRL|Linear|ReLU|Dropout|Linear] -> Domain loss
%  Backward: reversed gradient (GRL) feeds back to the pooled embedding.
% =====================================================================
\documentclass[border=15pt, multi, tikz]{standalone}
\usepackage{import}
\subimport{./layers/}{init}
\usetikzlibrary{positioning}
\usetikzlibrary{3d}

% ---- Named colors (work in Box fill=, \draw draw=, and \textcolor) ---
\colorlet{InputCol}{gray!20}
\colorlet{PearsonCol}{orange!40}
\colorlet{EdgeCol}{orange!65!yellow}
\colorlet{ConvCol}{blue!45!white}
\colorlet{BNCol}{teal!45!white}
\colorlet{ReLUCol}{orange!55!white}
\colorlet{PoolCol}{green!55!yellow}
\colorlet{LinearCol}{yellow!55!white}
\colorlet{DropCol}{red!30!white}
\colorlet{SAGECol}{blue!45!cyan}

\colorlet{LossCol}{gray!50}

\begin{document}
\begin{tikzpicture}

\node[above=0.3cm of input, align=center] {\textbf{Raw window}\\\scriptsize[60 $\times$ 6]};

\pic at (b1-east) {Box={name=r1,fill=ReLUCol,height=14,width=2,depth=14}};
% --- Conv1D block 2 : CNN1D | BN | ReLU
\pic[shift={(1.2,0,0)}] at (r1-east) {Box={name=c2,caption=Conv1D blk2,%
      fill=ConvCol,height=14,width=3,depth=14}};
\pic at (c2-east) {Box={name=b2,fill=BNCol,height=14,width=2,depth=14}};
\pic at (b2-east) {Box={name=r2,fill=ReLUCol,height=14,width=2,depth=14}};
% --- Adaptive Avg Pool
\pic[shift={(1.4,0,0)}] at (r2-east) {Box={name=ap,caption=AvgPool,%
      fill=PoolCol,height=12,width=3,depth=12}};
% --- Linear
\pic[shift={(1.4,0,0)}] at (ap-east) {Box={name=lin,caption=Linear,%
      fill=LinearCol,height=12,width=3,depth=12}};
% --- Node embeddings  (+ edge-type embed added here)
\pic[shift={(1.4,0,0)}] at (lin-east) {Box={name=nemb,caption=Node emb,%
      fill=LinearCol,height=10,width=3,depth=18}};

\pic[shift={(2.0,-3.2,0)}] at (input) {Box={name=prho,caption=Pearson $\rho$,%
      fill=PearsonCol,height=14,width=3,depth=14}};

      fill=PearsonCol,height=14,width=3,depth=14}};
\pic[shift={(1.6,0,0)}] at (score-east) {Box={name=topk,caption=Top-k,%
      fill=PearsonCol,height=14,width=3,depth=14}};
\pic[shift={(1.6,0,0)}] at (topk-east) {Box={name=eidx,caption=Edge\_index,%
      fill=EdgeCol,height=10,width=3,depth=18}};

% ===================== GraphSAGE MESSAGE PASSING ====================
% layer 1 : Dropout | SAGEConv | BN | ReLU
\pic[shift={(2.4,0,0)}] at (nemb-east) {Box={name=d1,caption=GraphSAGE $\times$2,%

\pic at (d1-east) {Box={name=s1,fill=SAGECol,height=16,width=3,depth=16}};
\pic at (s1-east) {Box={name=sb1,fill=BNCol,height=16,width=2,depth=16}};

\pic at (sr1-east){Box={name=d2,fill=DropCol,height=16,width=2,depth=16}};
\pic at (d2-east) {Box={name=s2,fill=SAGECol,height=16,width=3,depth=16}};

\pic[shift={(2.0,0,0)}] at (sr2-east) {Box={name=gpool,caption=Global\\Mean Pool,%

% ===================== ACTIVITY HEAD -> P_GNN ======================
\pic[shift={(2.2,2.6,0)}] at (gpool-east) {Box={name=alin,caption=Activity\\Linear,%

% GRL | Linear | ReLU | Dropout | Linear

\pic at (grl-east) {Box={name=dl1,fill=LinearCol,height=12,width=3,depth=12}};
\pic at (dl1-east) {Box={name=dre,fill=ReLUCol,height=12,width=2,depth=12}};
\pic at (dre-east) {Box={name=ddr,fill=DropCol,height=12,width=2,depth=12}};
\pic at (ddr-east) {Box={name=dl2,fill=LinearCol,height=12,width=3,depth=12}};
\pic[shift={(1.6,0,0)}] at (dl2-east) {Box={name=dloss,caption=Domain\\loss,%
      fill=LossCol,height=10,width=3,depth=10}};

% input -> both sub-paths
\draw[->,thick] (input) -- (c1-west);
\draw[->,thick] (input) -- (prho-west);

\draw[->,thick] (nemb-east) -- (d1-west);
\draw[->,thick] (eidx-east) -- (d1-west);

\draw[->,thick] (gpool-east) -- (alin-west);

\draw[->,thick] (smax-east) -- (pgnn-west);
